'use strict'; /* * Vercel Serverless Function: обслуживает ВСЕ запросы /api/*. * Путь подхватывается из file-based роутинга: api/[[...path]].js */ require('../lib/env').load(); const store = require('../lib/store'); const { apiHandler } = require('../lib/handler'); module.exports = async (req, res) => { try { await store.withRequest(() => apiHandler(req, res)); } catch (err) { console.error('[api]', err); if (!res.headersSent) { res.writeHead(500, { 'Content-Type': 'application/json; charset=utf-8' }); res.end(JSON.stringify({ error: 'Внутренняя ошибка сервера' })); } } }; // геокодирование + sms.ru иногда занимают больше 10 секунд module.exports.config = { maxDuration: 30 };
